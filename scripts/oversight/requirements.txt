# Directly imported by sign_off.sh and signoff_gate.py to parse the step
# manifest. PyYAML also arrives transitively via bandit/detect-secrets, but the
# gate must not depend on that luck — if a transitive provider drops it the
# oversight gate fails silently. Declare what we import. (CPS#73/#74, HOS#48/#49)
PyYAML>=6.0

radon>=5.1
bandit>=1.7
# semgrep and mcp are pinned EXACTLY and must match scripts/oversight/mcp_approved_list.json
# (#2197; the mcp_policy gate fails on any other version). semgrep 1.179.0 requires
# mcp==1.29.0. mcp is install-only: HOS must never import it or start semgrep's MCP server.
mcp==1.29.0
semgrep==1.179.0
detect-secrets>=1.4
pip-audit>=2.6

# Not imported by HOS (the App-token JWT is signed by openssl in
# bootstrap/get_app_token.sh). PyJWT arrives via semgrep and mcp; the floor exists
# so the shared oversight venv does not sit on 2.13.0's 13 advisories (#1776,
# ruling 2026-10-03). >=2.15.0 clears all 13 (PYSEC-2026-4146 affects only
# 2.11.0-2.13.0 per OSV), so no PyJWT advisory is exempted. semgrep 1.179.0
# (pinned above) requires pyjwt[crypto]>=2.15.0, so the pin and this floor agree.
pyjwt>=2.15.0

# Not imported by HOS. Arrives via pip-audit -> CacheControl[filecache], an
# extras-only edge, so `pip show filelock` reports Required-by: (none) and it
# looks orphaned; it is not. The floor exists so a stale venv re-synced with
# `pip install -r` drops 4.0.8's three advisories (#2164); >=4.2.1 clears all three.
filelock>=4.2.1

# Not imported by HOS; certifi arrives transitively via requests/httpx. The floor
# clears PYSEC-2026-4320 so security_scan does not depend on what the venv happens
# to have installed (#2147).
certifi>=2026.10.19

# PEP 440 ordering + PEP 503 canonicalisation for security_scan_logic.py (#1776).
# Arrives via pip-audit, but declare what we import (same rule as PyYAML above).
packaging>=23.0
# JS/TS AST pass (ADR-032 D4/D5/D8) — HOS-internal analysis, venv pip per D2
# (consumer-owned tools like eslint/tsc are discovered from the consumer
# instead, via resolve_node_tool.sh). Pinned to the tested-compatible pair:
# tree-sitter-typescript 0.23.x ships the capsule-based Language() API that
# tree-sitter >=0.22 expects.
tree-sitter>=0.23,<0.27
tree-sitter-typescript>=0.23,<0.24
# Formatters are version-PINNED to a major series: black's "stable style" changes
# by calendar-year major (24.x → 25.x → 26.x), so a floating `black>=24.0` let the
# venv install black 26 and reformat code the repo formatted with an older black —
# the lint gate then failed on pre-existing, untouched lines the author never wrote.
# Pin the major so the gate is deterministic across machines and over time. (#103)
flake8>=7.0,<8.0
flake8-bugbear>=24.0
black>=24.0,<25.0
isort>=5.13,<6.0
mypy>=1.8

# Testing
pytest>=8.0
pytest-cov>=5.0
# Retry tolerance for the required "tests" CI gate (#1244 ruling, 2026-09-10):
# a test that fails once then passes on an immediate retry is flaky by
# definition and must not block; a test that fails twice under identical
# conditions is a real failure and blocks, full stop. --reruns 1 in
# run_tests.sh implements the "2 total attempts" policy.
pytest-rerunfailures>=14.0
# mutmut is version-PINNED to a major series, same rationale as black above:
# 3.x rewrote its config format (paths_to_mutate/tests_dir/runner/dict_synonyms
# were dropped for source_paths/pytest_add_cli_args_test_selection, and the
# runner now shells pytest in-process instead of a --runner command). A
# floating `mutmut>=2.4` silently installed 3.6.0 against the 2.x-era
# pyproject.toml config, crashing every invocation. (#1083)
mutmut>=3.6,<4.0
