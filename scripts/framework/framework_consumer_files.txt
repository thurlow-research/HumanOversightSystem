# in bootstrap/hos_install.sh (mirrors consumer_agents.txt for agents). Edit here once;
# both the copy-loop and the manifest follow (#769).
#
# NOT listed here (deliberately):
#   - scripts/oversight/ — shipped via rsync, tracked via `find` in the manifest
#   - .claude/agents/ — governed by consumer_agents.txt
#   - .github/CODEOWNERS, .github/pull_request_template.md — handled by their own
#     installer section (also tracked directly in enumerate_framework_files)
#   - Internal HOS-dev tooling (must stay OUT of consumer installs):
#       cut_release.sh, validate_*.sh, check_agents_static.sh,
#       check_validation_current.sh, strip_internal_paths.sh,
#       run_framework_validation.sh, run_tests_release.sh
#

bootstrap/lib/branch_ownership.sh
bootstrap/create_branch.sh

# ── bootstrap/lib/ — sandbox-policy path helpers (#1221) ─────────────────────
# Sourced by hos_install.sh's --human sandbox-policy block. Guarded at the call
# site (declare -f check) so its absence degrades to a warn, not a hard-fail —
# but shipping it means fresh installs get it without a follow-up upgrade.
bootstrap/lib/sandbox_paths.sh

# ── .github/workflows/ — required CI status-check producers (#769) ───────────────
# These produce the status-check contexts that setup_branch_protection.sh requires.
# Without them a consumer's branch protection has required checks that never report,
# permanently blocking every PR.
.github/workflows/require-human-approval.yml
.github/workflows/require-overseer-approval.yml
.github/workflows/require-tier-ceiling.yml
.github/workflows/validation-check.yml

# ── .github/workflows/ — gate re-evaluation dispatcher (#1299) ───────────────────
# NOT itself a required status check (see test_branch_protection_contexts.py) — it
# only runs on pull_request_review events. The three gates above depend on it for
# review-triggered re-evaluation: without it, a gate that failed before an approval
# landed leaves a stale FAILURE check run that nothing re-evaluates.
.github/workflows/rerun-gate-checks.yml

scripts/framework/gen_codeowners.sh

# ── scripts/framework/ — required-check gate scripts (#1278) ─────────────────────
# require-human-approval.yml, require-overseer-approval.yml, and
# require-tier-ceiling.yml (already shipped above) invoke these. Without them the
# workflow runs, fails to find its script, and the required check can never pass.
scripts/framework/require_human_approval.py
scripts/framework/require_overseer_approval.py
scripts/framework/require_tier_ceiling.py

# ── scripts/framework/ — gate re-evaluation dispatcher script (#1299) ────────────
# Invoked by rerun-gate-checks.yml above. Reruns the three gates' EXISTING
# workflow runs in place on review submit/dismiss so a stale FAILURE check run
# doesn't block merge after an approval lands. Not itself a required check.
scripts/framework/rerun_gate_checks.py
